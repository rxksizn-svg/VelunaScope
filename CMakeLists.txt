cmake_minimum_required(VERSION 3.22)
project(VelunaScope VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
# Older dependency minimum versions remain valid with CMake 4.
set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(ENABLE_TESTS OFF CACHE BOOL "" FORCE)
include(FetchContent)
FetchContent_Declare(ebur GIT_REPOSITORY https://github.com/jiixyj/libebur128.git GIT_TAG v1.2.6 GIT_SHALLOW TRUE)
FetchContent_Declare(vad GIT_REPOSITORY https://github.com/dpirch/libfvad.git GIT_TAG v1.0 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(ebur vad)
file(GLOB_RECURSE VAD_SOURCES CONFIGURE_DEPENDS "${vad_SOURCE_DIR}/src/*.c")
add_library(VelunaVAD STATIC ${VAD_SOURCES})
target_include_directories(VelunaVAD PUBLIC "${vad_SOURCE_DIR}/include")
target_include_directories(ebur128 PUBLIC "${ebur_SOURCE_DIR}/ebur128")
include(CTest)
add_executable(ScopeTests Tests/ScopeTests.cpp)
target_include_directories(ScopeTests PRIVATE Source)
target_link_libraries(ScopeTests PRIVATE ebur128 VelunaVAD)
add_test(NAME ScopeMeasurements COMMAND ScopeTests)
option(VELUNA_METER_TESTS_ONLY "Build tests without JUCE" OFF)
if(VELUNA_METER_TESTS_ONLY)
  return()
endif()
FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.16-security GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(JUCE)
juce_add_binary_data(ScopeAssets SOURCES Assets/VelunaLogo.png)
juce_add_plugin(Scope COMPANY_NAME "Veluna Audio"
 PLUGIN_MANUFACTURER_CODE Vlna PLUGIN_CODE Vscp PRODUCT_NAME "Veluna Scope"
 FORMATS VST3 IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE
 COPY_PLUGIN_AFTER_BUILD FALSE VST3_CATEGORIES Fx Analyzer)
target_sources(Scope PRIVATE Source/Plugin.cpp Source/MeterCore.h)
target_include_directories(Scope PRIVATE Source)
target_compile_definitions(Scope PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(Scope PRIVATE ScopeAssets ebur128 VelunaVAD juce::juce_audio_utils
 PUBLIC juce::juce_recommended_config_flags)
